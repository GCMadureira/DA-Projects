38 int index;

40 std::unordered_set<int> selectedPallets;
41
42 std::string line;
43
44 //get the total profit on the first line
45 std::getline(output, line);
46 result.maxProfit = std::stoi(line);
47
48 //get the total weight on the second line
49 std::getline(output, line);
50 result.occupiedWeight = std::stoi(line);
51
52 //get the selected items' ids
53 std::getline(output, line);
54 std::istringstream iss(line);

59 // create the solution
