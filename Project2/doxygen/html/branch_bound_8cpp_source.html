14static void exploreLevel(int level, int currentValue, int currentWeight, int currentSize, int remainingValue) {
15 // reached the end of the tree OR the current pallet does not fit, so the following also do not
16 if(level == sortedPallets.size() || currentWeight + sortedPallets[level].weight > maxWeight) {
17 if(currentValue > bestValue || (currentValue == bestValue && currentSize < bestSize)
18 || (currentValue == bestValue && currentSize == bestSize && currentWeight < bestWeight)) {
19 //choose the solution based on profit, number of pallets and weight, in this order of priority
20 bestPacking = currentPacking;
21 bestValue = currentValue;
22 bestWeight = currentWeight;
23 bestSize = currentSize;
24 }
25 return;
26 }
27
28 const Pallet currentPallet = sortedPallets[level];
29
30 currentPacking[level] = true; // choose to add the current pallet if it fits
31 exploreLevel(level + 1, currentValue + currentPallet.profit, currentWeight + currentPallet.weight, currentSize + 1, remainingValue - currentPallet.profit);
32
33 currentPacking[level] = false; // choose not to add the current pallet
34 // if you choose not to add the pallet and it cannot lead to an optimal solution
35 if (currentValue + remainingValue - currentPallet.profit < bestValue) return;
36 exploreLevel(level + 1, currentValue, currentWeight, currentSize, remainingValue - currentPallet.profit);
37}

40KnapsackResult bb_approach(const ProblemInstance& instance) {
41 // set the static variables for this iteration of the algorithm
42 sortedPallets = instance.pallets;
43 currentPacking = std::vector<bool>(instance.pallets.size(), false);
44 bestPacking = std::vector<bool>(instance.pallets.size(), false);
45 bestValue = 0; bestWeight = INT_MAX; bestSize = 0;
46 maxWeight = instance.truckCapacity;
47
48 int remainingValue = 0;
49 for (Pallet pallet : instance.pallets) remainingValue += pallet.profit;
50
51 // sort the pallets according to their weights (asc)
52 std::sort(sortedPallets.begin(), sortedPallets.end(), [](const Pallet& left, const Pallet& right) {return left.weight < right.weight;});
53
54 exploreLevel(0, 0, 0, 0, remainingValue);
55
56 std::vector<Pallet> result;
57 int profit = 0, weight = 0;
58 for(int i = 0; i < sortedPallets.size(); i++) {
59 if(bestPacking[i]) {
60 result.push_back(sortedPallets[i]);
61 profit += sortedPallets[i].profit;
62 weight += sortedPallets[i].weight;
63 }
64 }
65
66 return {profit, weight, result};
