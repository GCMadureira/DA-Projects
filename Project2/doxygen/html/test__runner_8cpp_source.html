19double averageTime(std::function<void()> func) {
20 double totalTime = 0.0;
21 for (int i = 0; i < NUM_RUNS; ++i) {
22 auto start = std::chrono::high_resolution_clock::now();
23 func();
24 auto end = std::chrono::high_resolution_clock::now();
25 totalTime += std::chrono::duration<double>(end - start).count();
26 }
27 return totalTime / NUM_RUNS;
