16ProblemInstance readData(const std::string& truckFile, const std::string& palletFile) {
17 ProblemInstance instance;
18 std::ifstream truck(truckFile);
19 std::ifstream pallets(palletFile);
20
21 std::string line;
22
23 // Read truck capacity
24 getline(truck, line); // Skip header
25 if (!getline(truck, line)) {
26 std::cerr << "Failed to read truck data line.\n";
27 return instance;
28 }
29
30 // Remove carriage return if present
31 if (!line.empty() && line.back() == '\r')
32 line.pop_back();
33
34 std::stringstream ss(line);
35 std::string token;
36
37 // Capacity
38 getline(ss, token, ',');
39 token.erase(0, token.find_first_not_of(" \t")); // trim left
40 token.erase(token.find_last_not_of(" \t") + 1); // trim right
41 if (!parseInt(token, instance.truckCapacity)) {
42 std::cerr << "Failed to parse truck Capacity: " << token << "\n";
43 }
44
45 // Read Pallets
46 getline(pallets, line); // Skip header
47 while (getline(pallets, line)) {
48 if (line.empty()) continue;
49
50 if (!line.empty() && line.back() == '\r')
51 line.pop_back();
52
53 std::stringstream ps(line);
54 std::string token;
55 Pallet p;
56
57 // Pallet ID
58 getline(ps, token, ',');
59 if (!parseInt(token, p.id)) {
60 std::cerr << "Failed to parse ID: " << token << "\n";
61 continue;
62 }
63
64 // Weight
65 getline(ps, token, ',');
66 if (!parseInt(token, p.weight)) {
67 std::cerr << "Failed to parse weight: " << token << "\n";
68 continue;
69 }
70
71 // Profit
72 getline(ps, token, ',');
73 if (!parseInt(token, p.profit)) {
74 std::cerr << "Failed to parse profit: " << token << "\n";
75 continue;
76 }
77
78 instance.pallets.push_back(p);
79 }
80
81 return instance;
82}
