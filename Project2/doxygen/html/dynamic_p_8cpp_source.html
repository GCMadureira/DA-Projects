7KnapsackResult dp_array_approach(const ProblemInstance& instance) {
8 int capacity = instance.truckCapacity;
9 using State = std::pair<int, std::vector<int>>;
10
11 std::vector<State> dp(capacity + 1, {-1, {}});
12 dp[0] = {0, {}};
13
14 for (const auto& pallet : instance.pallets) {
15 for (int w = capacity - pallet.weight; w >= 0; --w) {
16 if (dp[w].first != -1) {
17 int newW = w + pallet.weight;
18 int newProfit = dp[w].first + pallet.profit;
19
20 if (dp[newW].first < newProfit ||
21 (dp[newW].first == newProfit && dp[newW].second.size() > dp[w].second.size() + 1)) {
22 dp[newW].first = newProfit;
23 dp[newW].second = dp[w].second;
24 dp[newW].second.push_back(pallet.id);
25 }
26 }
27 }
28 }
29
30 int maxProfit = 0;
31 std::vector<int> bestSelection;
32 for (int w = 0; w <= capacity; ++w) {
33 if (dp[w].first > maxProfit ||
34 (dp[w].first == maxProfit && dp[w].second.size() < bestSelection.size())) {
35 maxProfit = dp[w].first;
36 bestSelection = dp[w].second;
37 }
38 }
39
40 std::unordered_set<int> selectedPallets(bestSelection.begin(), bestSelection.end());
41 std::vector<Pallet> result;
42 int weight = 0;
43 for (const Pallet& pallet : instance.pallets) {
44 if (selectedPallets.contains(pallet.id)) {
45 result.push_back(pallet);
46 weight += pallet.weight;
47 }
48 }
49
50 return KnapsackResult{maxProfit, weight, result};
51}

53KnapsackResult dp_hashmap_approach(const ProblemInstance& instance) {
54 int capacity = instance.truckCapacity;
55 using State = std::pair<int, std::vector<int>>;
56
57 std::unordered_map<int, State> dp;
58 dp[0] = {0, {}};
59
60 for (const auto& pallet : instance.pallets) {
61 std::unordered_map<int, State> next = dp;
62
63 for (const auto& [weight, state] : dp) {
64 int newWeight = weight + pallet.weight;
65 if (newWeight > capacity) continue;
66
67 int newProfit = state.first + pallet.profit;
68
69 auto it = next.find(newWeight);
70 if (it == next.end() ||
71 it->second.first < newProfit ||
72 (it->second.first == newProfit && it->second.second.size() > state.second.size() + 1)) {
73 std::vector<int> newSelection = state.second;
74 newSelection.push_back(pallet.id);
75 next[newWeight] = {newProfit, std::move(newSelection)};
76 }
77 }
78
79 dp = std::move(next);
80 }
81
82 int maxProfit = 0;
83 std::vector<int> bestSelection;
84 for (const auto& [weight, state] : dp) {
85 if (state.first > maxProfit ||
86 (state.first == maxProfit && state.second.size() < bestSelection.size())) {
87 maxProfit = state.first;
88 bestSelection = state.second;
89 }
90 }
91
92 std::unordered_set<int> selectedPallets(bestSelection.begin(), bestSelection.end());
93 std::vector<Pallet> result;
94 int weight = 0;
95 for (const Pallet& pallet : instance.pallets) {
96 if (selectedPallets.contains(pallet.id)) {
97 result.push_back(pallet);
98 weight += pallet.weight;
99 }
100 }
101
102 return KnapsackResult{maxProfit, weight, result};
103}
