56void runInteractiveMode() {
57 std::string datasetPath = selectDatasetPath();
58 size_t sep = datasetPath.find(';');
59 std::string truckFile = datasetPath.substr(0, sep);
60 std::string palletFile = datasetPath.substr(sep + 1);
61
62 ProblemInstance instance = readData(truckFile, palletFile);
63
64 std::cout << "Dataset loaded. Pallets: " << instance.pallets.size()
65 << ", Capacity: " << instance.truckCapacity << "\n\n";
66
67 int choice;
68 do {
69 showMainMenu();
70 std::cin >> choice;
71
72 KnapsackResult result;
73
74 switch (choice) {
75 case 1:
76 result = bf_approach(instance);
77 break;
78 case 2:
79 result = bb_approach(instance);
80 break;
81 case 3:
82 result = dp_approach(instance);
83 break;
84 case 4: {
85 result = g_approach(instance);
86 // Compare greedy with optimal (DP)
87 int dpResult = dp_approach(instance).maxProfit;
88 if (result.maxProfit == dpResult)
89 std::cout << "\nGreedy solution is OPTIMAL.\n";
90 else
91 std::cout << "\nGreedy solution is NOT optimal -> " << result.maxProfit << "/" << dpResult << ", " << 100*result.maxProfit/(double)dpResult << "% of the optimal solution" << "\n";
92 break;
93 }
94 case 5: {
95 result = app_approach(instance);
96 // Compare approximation with optimal (DP)
97 int dpResult = dp_approach(instance).maxProfit;
98 if (result.maxProfit == dpResult)
99 std::cout << "\nApproximation solution is OPTIMAL.\n";
100 else
101 std::cout << "\nApproximation solution is NOT optimal -> " << result.maxProfit << "/" << dpResult << ", " << 100*result.maxProfit/(double)dpResult << "% of the optimal solution" << "\n";
102 break;
103 }
104 case 6:
105 result = ilp_approach(instance);
106 break;
107 case 0:
108 return;
109 default:
110 std::cout << "Invalid choice.\n";
111 continue;
112 }
113
114 std::sort(result.selectedPallets.begin(), result.selectedPallets.end(), [](Pallet left, Pallet right) {return left.id < right.id;});
115
116 std::cout << "\nMaximum profit: " << result.maxProfit << "\n";
117 std::cout << "Weight occupied: " << result.occupiedWeight << "/" << instance.truckCapacity << "\n";
118 std::cout << "Selected pallets (id:(profit,weight)):\n| ";
119 for (Pallet pallet : result.selectedPallets) {
120 std::cout << pallet.id << ":(" << pallet.profit << "," << pallet.weight << ") | "; ;
121 }
122 std::cout << "\n\n";
123
124 } while (choice != 0);
125}
