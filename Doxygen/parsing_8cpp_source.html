29std::vector<LocationRecord> readLocations(const std::string &filename) {
30 std::vector<LocationRecord> records;
31 std::ifstream file(filename);
32 if (!file.is_open()) {
33 std::cerr << "Error: cannot open file " << filename << std::endl;
34 return records;
35 }
36 std::string line;
37 std::getline(file, line); // Skip header line
38 while (std::getline(file, line)) {
39 //std::cout << "Reading line: " << line << std::endl;
40 std::istringstream linestream(line);
41 std::string token;
42 LocationRecord rec;
43 // Field 1: Location (name)
44 std::getline(linestream, token, ',');
45 rec.location = token;
46 // Field 2: Id
47 std::getline(linestream, token, ',');
48 rec.id = std::stoi(token);
49 // Field 3: Code
50 std::getline(linestream, token, ',');
51 rec.code = token;
52 // Field 4: Parking (1 or 0)
53 std::getline(linestream, token, ',');
54 rec.parking = stringToBool(token);
55 records.push_back(rec);
56 }
57 file.close();
58 return records;
59}

71std::vector<DistanceRecord> readDistances(const std::string &filename) {
72 std::vector<DistanceRecord> records;
73 std::ifstream file(filename);
74 if (!file.is_open()) {
75 std::cerr << "Error: cannot open file " << filename << std::endl;
76 return records;
77 }
78 std::string line;
79 std::getline(file, line); // Skip header line
80 while (std::getline(file, line)) {
81 //std::cout << "Reading line: " << line << std::endl;
82 std::istringstream linestream(line);
83 std::string token;
84 DistanceRecord rec;
85 // Field 1: Location1 (start location code)
86 std::getline(linestream, token, ',');
87 rec.code1 = token;
88 // Field 2: Location2 (end location code)
89 std::getline(linestream, token, ',');
90 rec.code2 = token;
91 // Field 3: Driving (if "X", set drivingTime to -1)
92 std::getline(linestream, token, ',');
93 if (token == "X")
94 rec.drivingTime = -1; // Used -1 to denote an undrivable segment
95 else
96 rec.drivingTime = std::stoi(token);
97 // Field 4: Walking
98 std::getline(linestream, token, ',');
99 rec.walkingTime = std::stoi(token);
100 records.push_back(rec);
101 }
102 file.close();
103 return records;
104}

114void loadGraph(Graph<int>& graph) {
115 std::vector<LocationRecord> locations = readLocations("DataSets/Locations2.csv");
116 std::vector<DistanceRecord> distances = readDistances("DataSets/Distances2.csv");
117
118 std::unordered_map<std::string, int> codeToId;
119
120
121 // Add locations (nodes) to the graph
122 for (const auto& loc : locations) {
123 graph.addVertex(loc.id);
124 codeToId[loc.code] = loc.id;
125 if(loc.parking){
126 graph.findVertex(loc.id)->setParking(true);
127 }
128 }
129
130 // Add distances (edges) to the graph
131 for (const auto& dist : distances) {
132 if (codeToId.find(dist.code1) != codeToId.end() && codeToId.find(dist.code2) != codeToId.end()) {
133 int id1 = codeToId[dist.code1];
134 int id2 = codeToId[dist.code2];
135
136 graph.addBidirectionalEdge(id1, id2, dist.drivingTime, dist.walkingTime);
137 } else {
138 std::cerr << "Warning: Skipping edge (" << dist.code1 << ", " << dist.code2 << ") due to missing location.\n";
139 }
140 }
141
142 std::cout << "Graph successfully loaded with " << locations.size() << " locations and " << distances.size() << " connections.\n";
143}

180int parseSource(const Graph<int>& graph, const std::string& sourceLine) {
181 int startNode;
182 size_t pos = sourceLine.find(':');
183 if (pos != std::string::npos) {
184 std::string startNodeStr = sourceLine.substr(pos + 1);
185 try {startNode = std::stoi(startNodeStr);}
186 catch (const std::exception& e) {
187 std::cout << "Invalid node format. Source node must be a valid integer.\n";
188 return -1;
189 }
190 if (graph.findVertex(startNode) == nullptr) {
191 std::cout << "The source node " << startNode << " does not exist.\n";
192 return -1;
193 }
194 } else {
195 std::cout << "Invalid input format. Expected 'Source:<id>'\n";
196 return -1;
197 }
198 return startNode;

312void parseAvoidSegments(const Graph<int>& graph, const std::string& avoidSegmentsLine) {
313 std::string avoidSegmentsStr;
314 size_t pos = avoidSegmentsLine.find(':');
315 if (pos != std::string::npos) avoidSegmentsStr = avoidSegmentsLine.substr(pos + 1);
316 std::stringstream segSS(avoidSegmentsStr);
317 std::string segmentStr;
318 while (std::getline(segSS, segmentStr, ')')) {
319 size_t openParen = segmentStr.find('(');
320 size_t comma = segmentStr.find(',', 1);
321 if (openParen != std::string::npos && comma != std::string::npos) {
322 try {
323 int from = std::stoi(segmentStr.substr(openParen + 1, comma - openParen - 1));
324 int to = std::stoi(segmentStr.substr(comma + 1));
325 if (auto edge = graph.findEdge(from, to)) { // Check if edge is not nullptr
326 edge->setIgnoreFlag(true);
327 } else {
328 std::cout << "Warning: Edge (" << from << ", " << to << ") not found in graph.\n";
329 }
330 } catch (const std::exception&) {
331 std::cout << "Warning: Invalid edge format in AvoidSegments.\n";
332 }
333 }
334 else std::cout << "Warning: Invalid edge format in AvoidSegments.\n";
335 }
336}
