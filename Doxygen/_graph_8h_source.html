174template <class T>
175bool Vertex<T>::removeEdge(T in) {
176 bool removedEdge = false;
177 auto it = adj.begin();
178 while (it != adj.end()) {
179 Edge<T> *edge = *it;
180 Vertex<T> *dest = edge->getDest();
181 if (dest->getInfo() == in) {
182 it = adj.erase(it);
183 deleteEdge(edge);
184 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
185 }
186 else {
187 it++;
188 }
189 }
190 return removedEdge;
191}

197template <class T>
198void Vertex<T>::removeOutgoingEdges() {
199 auto it = adj.begin();
200 while (it != adj.end()) {
201 Edge<T> *edge = *it;
202 it = adj.erase(it);
203 deleteEdge(edge);
204 }
205}

558template <class T>
559bool Graph<T>::removeVertex(const T &in) {
560 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
561 if ((*it)->getInfo() == in) {
562 auto v = *it;
563 v->removeOutgoingEdges();
564 for (auto u : vertexSet) {
565 u->removeEdge(v->getInfo());
566 }
567 vertexSet.erase(it);
568 delete v;
569 return true;
570 }
571 }
572 return false;
573}

620template <class T>
621Edge<T>* Graph<T>::findEdge(const T &sourc, const T &dest) const{
622 Vertex<T>* srcVertex = findVertex(sourc);
623 if (srcVertex == nullptr) return nullptr;
624 for (auto e : srcVertex->getAdj()) {
625 if (e->getDest()->getInfo() == dest)
626 return e;

641template <class T>
642bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, int drivingTime, int walkingTime) {
643 auto v1 = findVertex(sourc);
644 auto v2 = findVertex(dest);
645 if (v1 == nullptr || v2 == nullptr)
646 return false;
647
648 // Add edges in both directions
649 auto e1 = v1->addEdge(v2, drivingTime, walkingTime);
650 auto e2 = v2->addEdge(v1, drivingTime, walkingTime);
651
652 e1->setReverse(e2);
653 e2->setReverse(e1);
654
655 return true;
656}
