9template <class T>
10bool relax(Edge<T> *edge, const bool driving) { // d[u] + w(u,v) < d[v]
11 Vertex<T>* neighbor = edge->getDest();
12 Vertex<T>* v = edge->getOrig();
13 if(driving) {
14 if ((v->getDist() + edge->getDrivingTime()) < neighbor->getDist()) {
15 neighbor->setDist(v->getDist() + edge->getDrivingTime());
16 neighbor->setPath(edge);
17 return true;
18 }
19 }
20 else{
21 if ((v->getDist() + edge->getWalkingTime()) < neighbor->getDist()) {
22 neighbor->setDist(v->getDist() + edge->getWalkingTime());
23 neighbor->setPath(edge);
24 return true;
25 }
26 }
27 return false;
28}

30template <class T>
31void dijkstra(Graph<T> * g, const int &origin, const bool driving) {
32 if (!g->findVertex(origin)) return;
33
34 std::vector<Vertex<T>*> vertices = g->getVertexSet();
35 MutablePriorityQueue<Vertex<T>> pq;
36 for(Vertex<T>* v : vertices){
37 v->setDist(INF);
38 v->setPath(nullptr);
39 v->setVisited(false);
40 }
41
42 g->findVertex(origin)->setDist(0);
43 pq.insert(g->findVertex(origin));
44
45 while(!pq.empty()){
46 Vertex<T>* v = pq.extractMin();
47 v->setVisited(true);
48 for(Edge<T>* e : v->getAdj()){
49 if (e->getDest()->isIgnored() || e->isIgnored() || e->getDest()->isVisited()) continue; // Restricted Route or already visited
50 if (e->getDest()->getDist() == INF) pq.insert(e->getDest());
51 if(driving) {
52 if (e->getDrivingTime() != -1 && relax(e)) { //For Driving route
53 pq.decreaseKey(e->getDest());
54 }
55 }
56 else{
57 if (e->getWalkingTime() != -1 && relax(e,false)) { //For Walking Route
58 pq.decreaseKey(e->getDest());
59 }
60 }
61 }
62 }
63}

65template <class T>
66std::vector<T> getPath(Graph<T> * g, const int &origin, const int &dest, const bool savedPath) {
67 std::vector<T> path;
68 Vertex<T>* v = g->findVertex(dest);
69
70 // if normal path's destination vertex does not exist or its dist is undefined return
71 if (!savedPath && (!v || v->getDist() == INT_MAX)) return path;
72
73 // if saved path's destination vertex does not exist or its savedDist is undefined return
74 if (savedPath && (!v || v->getSavedDist() == INT_MAX)) return path;
75
76 while (v) {
77 path.push_back(v->getInfo());
78 if (v->getInfo() == origin) break;
79
80 if (!savedPath) v = v->getPath()->getOrig();
81 else v = v->getSavedPath()->getOrig();
82 }
83
84 reverse(path.begin(), path.end());
85 return path;
